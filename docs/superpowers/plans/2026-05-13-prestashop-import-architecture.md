# PrestaShop Import Architecture Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expand the PrestaShop import orchestrator to support complex stateful relational data including taxes, categories, combinations with impact prices, native stock updates, and password generation.

**Architecture:** A stateful frontend orchestrator that extracts and creates parent entities (categories, taxes, attributes) via PrestaShop's API, caching their IDs and properties in memory to eliminate redundant GET requests and ensure accurate calculations for child entities (products, combinations, orders).

**Tech Stack:** TypeScript, Vue 3, Vite, Axios, xml2js.

function generateSecurePassword(): string {
    // Basic fallback if crypto is not available, but generates a solid random string
    const array = new Uint32Array(4);
    window.crypto.getRandomValues(array);
    return Array.from(array, dec => dec.toString(16).padStart(8, "0")).join("") + 'A1!';
}

export function buildCustomerXml(data: Record<string, string>): string {
    if (!data.firstname || data.firstname.trim() === '') data.firstname = 'Inconnu';
    if (!data.lastname || data.lastname.trim() === '') data.lastname = 'Inconnu';

    const customer: Record<string, string> = {
        active: data.active || '1',
        id_default_group: data.id_default_group || '3',
        newsletter: data.newsletter || '0',
        firstname: data.firstname,
        lastname: data.lastname,
        email: data.email || '',
        passwd: data.passwd || generateSecurePassword(),
    };

    return buildXml('customer', Object.entries(customer));
}

interface TaxCacheInfo {

// Helper to parse tax string to float
function parseTaxRate(taxString: string): number {
    if (!taxString) return 0;
    const clean = taxString.replace('%', '').replace(',', '.').trim();
    const rate = parseFloat(clean);
    return isNaN(rate) ? 0 : rate;
}

// Insert before Phase 3 (Produits) in orchestratorImport
// Phase 1.5 : Taxes
if (productFiles.length > 0) {
    reportProgress('mapping', 'Création des taxes');
    const allProductRows = productFiles.flatMap(f => f.rows);
    const uniqueTaxes = new Set<string>();
    
    // Config key for tax column is 'Taxe', 'taxe', etc. We'll look for raw keys mapped to id_tax_rules_group
    allProductRows.forEach(row => {
        const taxVal = row['Taxe'] || row['taxe'] || row['tax'];
        if (taxVal && taxVal.trim() !== '') uniqueTaxes.add(taxVal.trim());
    });

    for (const taxStr of uniqueTaxes) {
        if (taxCache.has(taxStr)) continue;
        
        const rate = parseTaxRate(taxStr);
        try {
            // Create tax_rules_group
            const trgXml = buildXml('tax_rules_group', [['name', `Taxe ${rate}%`], ['active', '1']]);
            const trgRes = await apiService.post<any>('/tax_rules_groups', trgXml, { headers: { 'Content-Type': 'application/xml' } });
            const trgId = parseInt(trgRes?.prestashop?.tax_rules_group?.id || '0', 10);

            // Create tax
            const tXml = buildXml('tax', [['name', `Tax ${rate}%`], ['rate', String(rate)], ['active', '1']]);
            const tRes = await apiService.post<any>('/taxes', tXml, { headers: { 'Content-Type': 'application/xml' } });
            const tId = parseInt(tRes?.prestashop?.tax?.id || '0', 10);

            // Create tax_rule
            if (trgId > 0 && tId > 0) {
                const trXml = buildXml('tax_rule', [
                    ['id_tax_rules_group', String(trgId)],
                    ['id_tax', String(tId)],
                    ['id_country', '0'],
                    ['id_state', '0'],
                    ['id_county', '0'],
                    ['behavior', '0']
                ]);
                await apiService.post<any>('/tax_rules', trXml, { headers: { 'Content-Type': 'application/xml' } });
                taxCache.set(taxStr, { id_tax_rules_group: trgId, rate });
            }
        } catch (e) {
            console.error(`Erreur création taxe ${taxStr}`, e);
        }
    }
}
